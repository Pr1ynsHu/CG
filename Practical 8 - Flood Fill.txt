/* Practical VIII - Flood fill algorithm (4-connected)
   The seed point must lie in the region having oldColor.
   Borland Turbo C++ / BGI
*/
#include <graphics.h>
#include <conio.h>

void floodFill(int x, int y, int fillColor, int oldColor)
{
    if (x < 0 || x > getmaxx() || y < 0 || y > getmaxy())
        return;

    if (getpixel(x, y) != oldColor)
        return;

    putpixel(x, y, fillColor);

    floodFill(x + 1, y, fillColor, oldColor);
    floodFill(x, y - 1, fillColor, oldColor);
    floodFill(x - 1, y, fillColor, oldColor);
    floodFill(x, y + 1, fillColor, oldColor);
}

int main(void)
{
    int gd = DETECT, gm;

    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");
    if (graphresult() != grOk)
        return 1;

    setbkcolor(WHITE);
    cleardevice();
    setcolor(BLUE);
    rectangle(100, 100, 150, 150);

    /* BGI background is BLACK; the seed is inside the rectangle. */
    floodFill(101, 101, CYAN, BLACK);

    getch();
    closegraph();
    return 0;
}
