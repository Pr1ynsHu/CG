/*------------------- 1.1 To draw Pixels -------------------*/
#include <conio.h>
#include <graphics.h>

void main(){
    // Initialize graphics library
    int gd = DETECT, gm, i, j;
    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    // Set bgcolor to white
    setbkcolor(15);

    // Draw 4 pixels on the screen
    putpixel(25, 25, RED);
    putpixel(30, 25, RED);
    putpixel(35, 25, RED);
    putpixel(40, 25, RED);

    // Closing statements
    getch();
    closegraph();
}

/*------------------- 1.2 To draw a Line -------------------*/
#include <conio.h>
#include <graphics.h>

void main(){
    // Initialize graphics library
    int gd = DETECT, gm, i, j;
    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    // Set bgcolor to white
    setbkcolor(WHITE);

    // Draw a line from (25,25) to (150,150)
    for(int x = 0; x < 125; x++){
        putpixel(25 + x, 25 + x, RED);
    }

    getch();
    closegraph();
}

/*------------------- 1.3 To draw a Rectangle -------------------*/
#include <conio.h>
#include <graphics.h>

void main(){
    int gd = DETECT, gm, i, j;
    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    setbkcolor(WHITE);

    // draw top edge
    for(int topx = 25; topx <= 150; topx++){
        putpixel(topx, 25, RED);
    }
    // draw bottom edge
    for(int bottomx = 25; bottomx <= 150; bottomx++){
        putpixel(bottomx, 225, RED);
    }
    // draw left edge
    for(int lefty = 25; lefty <= 225; lefty++){
        putpixel(25, lefty, RED);
    }
    // draw right edge
    for(int righty = 25; righty <= 225; righty++){
        putpixel(150, righty, RED);
    }

    getch();
    closegraph();
}
